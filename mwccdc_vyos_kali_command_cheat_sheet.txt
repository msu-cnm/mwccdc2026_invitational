--- 1A. OPERATIONAL MODE (Diagnostics & Threat Hunting) ---

# System, Routing & Interface Checks

show configuration commands                         # Dump full running config as 'set' commands

# Active Threat Hunting & Traffic Inspection
show conntrack table ipv4                           # View live stateful/NAT sessions
show conntrack table ipv4 | grep <SUSPICIOUS_IP>    # Filter live connections by IP
show log tail 50                                    # View last 50 lines of system logs
show log auth                                       # Check for SSH brute-force / login attempts
monitor traffic interface eth0                      # Live packet capture on interface
monitor traffic interface eth0 filter 'not port 22' # Capture traffic excluding SSH

# User & OS-Level Persistence Auditing
show users                                          # Show currently logged-in sessions
show configuration commands | grep user             # Audit configured VyOS users & SSH keys
sudo ss -tulnp                                      # Check listening sockets on underlying Debian OS
sudo crontab -l && ls -la /etc/cron*                # Check for unauthorized cron jobs

--- 1B. CONFIGURATION MODE (Hardening & Safety Nets) ---
Note: Type 'configure' to enter. Changes require 'commit' to apply and 'save' to persist.

confirm                                             # Run within 5 mins to keep changes permanent
compare                                             # Show diff between candidate and active config
rollback 1                                          # Revert to previous saved config revision

# Immediate First-5-Minute Hardening

commit-confirm 5
confirm

# Blocking a Malicious Red Team IP (VyOS 1.4 nftables Syntax)
configure
set firewall group address-group RED_TEAM_BLOCK address '<MALICIOUS_IP>'

# Apply block to traffic passing THROUGH VyOS (Forward chain)

set firewall ipv4 forward filter rule 10 action drop
set firewall ipv4 forward filter rule 10 source group address-group RED_TEAM_BLOCK

# Apply block to traffic destined FOR VyOS (Input chain)
set firewall ipv4 input filter default-action accept
set firewall ipv4 input filter rule 10 action drop
set firewall ipv4 input filter rule 10 source group address-group RED_TEAM_BLOCK
commit
save

# Adding Subsequent Bad IPs Later:

--- 2B. NETWORK MONITORING & THREAT HUNTING ---

# Live Packet Capture (tcpdump)
sudo tcpdump -i eth0 -nn 'not port 22'              # Watch live traffic excluding SSH
sudo tcpdump -i eth0 -nn -w /tmp/cap.pcap -C 50 -W 5 # Rotating 50MB PCAPs (keeps last 5)

# Protocol Triage (tshark)
sudo tshark -i eth0 -Y "http.request" -T fields -e ip.src -e http.host -e http.request.uri
sudo tshark -i eth0 -Y "dns.flags.response == 0" -T fields -e ip.src -e dns.qry.name

# Stream Remote Packet Capture from VyOS Directly into Kali Wireshark
ssh vyos@172.17.0.1 "sudo tcpdump -i eth0 -U -w - 'not port 22'" | wireshark -k -i -
